# ============================================================================
#  DUI Dynamic Topbar HUD  --  开局初始化
#  ---------------------------------------------------------------------------
#  on_actions 的同名块在引擎里是「追加合并」的（原版 00_on_actions.txt 自己
#  就写了两遍 on_startup），所以这里直接追加不会覆盖原版内容。
#
#  每个国家开局写入默认布局，保证一进游戏就能看到信息条。
#  即使这个文件因为任何原因没生效，玩家点一次「+」按钮也会触发兜底初始化。
# ============================================================================

on_actions = {

	on_startup = {
		effect = {
			every_country = {
				dui_topbar_init_if_needed = yes
			}
		}
	}

	# 注意：稳定度 / 战争支持度的「每周变化」现在直接读原版修饰符
	#       modifier@stability_weekly / modifier@war_support_weekly（和悬停窗口一致），
	on_daily = {
		effect = {
			every_country = { dui_topbar_equip_snapshot = yes }
		}
	}
}
