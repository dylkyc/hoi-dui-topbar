#      Shift+右键"+" -> 恢复默认布局

	# ========================================================================
	#  指标选择菜单
	# ========================================================================
	dui_topbar_menu = {

		context_type = player_context

		window_name = "dui_menu_panel"
		parent_window_token = top_bar

		# 菜单关闭时整个窗口隐藏，菜单关闭时数组也已清空
		visible = {
			has_variable = dui_menu_open
		}

		# 只有菜单内容变化时才重建候选列表（脚本化效果里 add_to_variable dui_menu_dirty）
		dirty = dui_menu_dirty

		properties = {
			dui_menu_item_icon = { image = "[GetDUIMenuItemIcon]" }
		}

		dynamic_lists = {
			dui_menu_grid = {
				array = Root.dui_menu_items
				value = dui_menu_v
				index = dui_menu_i
				change_scope = no
				entry_container = "dui_menu_entry"
			}
		}


			triggers = {
				dui_menu_expand_visible = {
					is_in_array = { dui_subcategory = dui_menu_v }
				}
			}

		effects = {

			# 选中某个指标
			dui_menu_item_click = {
			if = {
				limit = { has_variable = dui_sub_open }
				dui_topbar_assign = yes
				dui_topbar_close_sub = yes
			}
			else = {
				if = {
					limit = { is_in_array = { dui_subcategory = dui_menu_v } }
					dui_topbar_fill_sub = yes
				}
				else = {
					dui_topbar_assign = yes
				}
			}
		}
			# Shift + 左键：选中并再追加一个空格子

			dui_menu_item_shift_click = {
				meta_effect = {
					text = {
						custom_effect_tooltip = "[X]_tt"
					}
					X = "[?dui_menu_v.GetTokenKey]"
				}
				custom_effect_tooltip = DUI_MENU_ITEM_HELP_SHIFT
				dui_topbar_assign_append = yes
			}
			# 关闭菜单
			# 弹窗里的「+」按钮：左键加格子 / 右键折叠 / Shift+右键复位
			dui_menu_add_click = {

			}

			dui_menu_close_click = {
				dui_topbar_close_menu = yes
			}
		}
